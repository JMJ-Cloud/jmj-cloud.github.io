SELECT  s.vendor_name                          supplier
,       SUM(i.invoice_amount - i.amount_paid)  open_amt
FROM    ap_invoices_all  i
,       ap_suppliers     s
WHERE   s.vendor_id            = i.vendor_id
AND     i.payment_status_flag  = 'N'
AND     i.invoice_date         >= TRUNC(SYSDATE,'Q')
GROUP BY s.vendor_name
ORDER BY open_amt DESC
